game.screen_manager

Gestionnaire d'ecrans (ScreenManager).

Role :

  • Connaitre l'etat global de l'application (titre / jeu / pause / inventaire / game over / parametres).
  • Afficher ou masquer les ecrans sur la scene courante.
  • Router les evenements clavier et souris vers l'ecran actif.
  • Creer une nouvelle GameScene propre lors d'un reset ou d'un retour au menu.
  • Gerer la pause et l'inventaire

 58    def _create_fresh_scene(self):
 59        """Arrete l'ancienne scene, en cree une nouvelle et la retourne."""
 60        from game.scene import GameScene   # import tardif pour eviter les imports circulaires
 61
 62        old = self._scene
 63        if old is not None:
 64            if hasattr(old, 'timer'):
 65                old.timer.stop()
 66
 67        new_scene = GameScene(screen_manager=self)

 69        self.window.setScene(new_scene)

100    def route_key_press(self, key):
101
102        """ Transmet la touche a l'ecran actif s'il y en a un.
103        Sinon, intercepte Echap (→ pause) et Tab (→ inventaire) pendant le jeu.
104        Retourne True si l'evenement est consomme (ne doit pas atteindre le jeu)."""
105        if self._active_screen is not None:
106            self._active_screen.key_press(key)
107            return True
108
109        if self.state == self.STATE_GAME:
110            if self._scene and self._scene.dialogue_manager.active:
111                # on ne fait rien, on laisse la touche passer au jeu mais empeche ouverture
112                if key == settings.keys["PAUSE"] or key == settings.keys["INVENTORY"]:
113                    return True
114            if key == settings.keys["PAUSE"]:
115                self.open_pause()
116                return True
117            if key == settings.keys["INVENTORY"]:
118                self.open_inventory()
119                return True
120
121        return False
122
123    def route_key_release(self, key):
124        """Transmet le key_release a l'ecran actif (pour l'etat 'pressed' des boutons)."""
125        if self._active_screen is not None:
126            self._active_screen.key_release(key)
127            return True
128        return False

144
145        """Retourne a l'ecran titre.
146        Cree une scene fraiche (en pause) et affiche le menu principal. """
147        self.hide_current_screen()

152

156    def go_to_settings(self):
157        """Affiche l'ecran des parametres en memorisant l'etat precedent."""
158        # verifie si on vient de l'ecran titre pour autoriser le re-scale
159        allow_rescale = (self.state == self.STATE_TITLE)
160        
161        settings_scr = self._screens.get("settings")
162        if settings_scr:
163            settings_scr.set_rescale_allowed(allow_rescale)
164        
165        self._prev_state = self.state
166        self.show_screen("settings")
167        self.state = self.STATE_SETTINGS
168
169    def back_from_settings(self):
170        """ Retourne a l'ecran precedent depuis les parametres.
171          - Depuis le titre   : reaffiche le titre.
172          - Depuis le jeu     : reprend le jeu.
173          - Depuis la pause   : reaffiche le menu pause."""
174        prev = self._prev_state or self.STATE_TITLE
175        if prev == self.STATE_TITLE:
176            self.show_screen("title")
177            self.state = self.STATE_TITLE
178        elif prev == self.STATE_GAME:
179            self.hide_current_screen()
180            self.state = self.STATE_GAME
181        elif prev == self.STATE_PAUSED:
182            self.show_screen("pause")
183            self.state = self.STATE_PAUSED
184        else:
185            self.show_screen("title")
186            self.state = self.STATE_TITLE

208    def open_pause(self):
209        """Met le jeu en pause et affiche le menu correspondant."""
210        if self.scene is None:
211            return
212        if hasattr(self.scene, 'player'):
213            self.scene.player.stop_movement()
214        
215        self.scene.game_paused = True
216        if hasattr(self.scene, 'music_manager'):
217            self.music_manager.stop()
218        if hasattr(self._scene, 'sfx_manager'):
219            self.scene.sfx_manager.stop_all_except()
220            self.scene.sfx_manager.play("snd_sys_pause")
221        # fonction labma pour check si on est toujours en pause avant de jouer (empeche spam)
222        QTimer.singleShot(200, lambda: self._play_music_if_state("mus_pause", self.STATE_PAUSED))
223            
224        self.show_screen("pause")
225        self.state = self.STATE_PAUSED
226
227    def resume_game(self):
228        """Reprend le jeu et relance la musique de la room avec fade-in."""
229        self.hide_current_screen()
230        if self._scene is not None:
231            if hasattr(self._scene, 'sfx_manager'):
232                self._scene.sfx_manager.play("snd_sys_resume")
233
234            self._scene.start_room_music()
235            self._scene.game_paused = False
236            
237        self.state = self.STATE_GAME

273    def toggle_inventory(self):
274        """Ouvre ou ferme l'inventaire selon l'etat actuel."""
275        if self.state == self.STATE_INVENTORY:
276            self.close_inventory()
277        elif self.state == self.STATE_GAME:
278            self.open_inventory()

280    def _play_music_if_state(self, music_name, target_state):
281        """Joue la musique uniquement si le manager est encore dans l'etat voulu."""
282        if self.state == target_state and self._scene and hasattr(self._scene, 'music_manager'):
283            self.music_manager.play(music_name)

302    def start_new_game(self):
303        """Initialise une nouvelle partie et affiche l'ecran des controles."""
304    
305        self.hide_current_screen()
306    
307        scene = self._create_fresh_scene()
308    
309        # save temporaire non liée à un slot
310        scene.current_save = SaveManager(slot=None)
311    
312        scene.load_current_save()
313    
314        if self.settings is not None:
315            self.settings.apply_to_scene(scene)
316    
317        if self.inventory is not None:
318            self.inventory.reset()
319            # # items de depart (hors debug)
320            # self.inventory.add_item("bombe", 5)
321            # self.inventory.add_item("pomme", 5)
322            # self.inventory.equip_item("bombe")
323        if hasattr(self, 'music_manager'):
324            self.music_manager.stop()
325            
326        controls_scr = self._screens.get(self.STATE_CONTROLS)
327        if hasattr(controls_scr, 'reset_build'):
328            controls_scr.reset_build()
329            
330        self.show_screen("controls")
331        self.state = self.STATE_CONTROLS
332

334    def finalize_new_game(self, wipe_item):
335        """Affiche la scene de jeu avec un effet de transition."""
336        
337        if self._scene is not None:
338            self._scene.game_paused = False
339            self._scene.start_room_music()
340            
341            # detacher overlay de l'ecran de controle pour le mettre a la racine 
342            # et gerer separement pour fondu d'ouverture
343            if self._active_screen and wipe_item in self._active_screen._items:
344                self._active_screen._items.remove(wipe_item)
345            
346            self._scene.addItem(wipe_item)
347        
348        self.state = self.STATE_GAME
349        self.hide_current_screen() 
350        
351        # lancer fondu overlay
352        self.final_fade = QPropertyAnimation(wipe_item, b"opacity")
353        self.final_fade.setDuration(500) # 0.2s
354        self.final_fade.setStartValue(1.0)
355        self.final_fade.setEndValue(0.0)
356        
357        # retire de la scene
358        self.final_fade.finished.connect(lambda: self._scene.removeItem(wipe_item) if self._scene else None)
359        self.final_fade.start()
360

400    def rebuild_display(self):
401        """Reconstruit tout l'affichage lors d'un changement de resolution."""
402        # cacher ecran actuel
403        self.hide_current_screen()
404        
405        # vider tous les ecran pour re forcer _build
406        for screen in self._screens.values():
407            if hasattr(screen, 'reset_build'):
408                screen.reset_build()
409                
410        # change taille fenetre
411        if hasattr(self.window, 'update_window_size'):
412            self.window.update_window_size()
413            
414        # re-cree scene vierge
415        self._create_fresh_scene()
416        
417        # re-affiche ecran de parametres
418        self.show_screen("settings")
