396    def handle_inputs(self, dt, scene):
397        """Gere les entrees clavier (deplacement, attaque, projectiles)."""
398        
399        #bloque mouvements du joueur si dialogue
400        if scene.dialogue_manager.active or self.is_cinematic:
401            self.attack_pressed = True
402            return
403        
404        dx, dy = 0, 0
405        if not self.kb_active:
406            if settings.keys["UP"] in self.keys:    dy -= 1; self.direction = "up"
407            if settings.keys["DOWN"] in self.keys:  dy += 1; self.direction = "down"
408            if settings.keys["LEFT"] in self.keys:  dx -= 1; self.direction = "left"
409            if settings.keys["RIGHT"] in self.keys: dx += 1; self.direction = "right"
410
411         # normalisation diagonale
412        if dx != 0 and dy != 0:
413            dx *= 0.707106
414            dy *= 0.707106
415
416        self.move(dx, dy, dt, scene)
417
418        if settings.keys["ATTACK"] in self.keys:
419            if not self.attack_pressed and self.attack_cooldown == 0:
420                self.attack(scene)
421                self.attack_pressed = True
422                self.attack_cooldown = self.attack_delay
423        elif settings.keys["ITEM"] in self.keys:
424            if not self.attack_pressed and self.projectiles_cooldown == 0:
425                if use_item(self, scene):
426                    self.attack_pressed = True
427                    self.projectiles_cooldown = self.projectiles_delay
428                else:
429                    self.attack_pressed = True
430        else:
431            self.attack_pressed = False
432            
433        if settings.keys["SHOUTS"] in self.keys:
434            if not self.shout_pressed and self.shout_cooldown == 0:
435                self.shout(scene)
436                self.shout_pressed = True
437                self.shout_cooldown = self.shout_delay
438        else:
439            self.shout_pressed = False
440        
441        
442        if DEBUG:
443            if settings.keys["CROUCH"] in self.keys:
444                self.debug_speed_multiplier = 0.5
