 141    def game_loop(self):
 142        """Boucle principale declenchee a chaque frame (mise a jour entites, HUD, salle)."""
 143        # last_time est mis a jour en premier pour eviter un grand dt au depause
 144        current_time = time.time()
 145        dt = min(current_time - self.last_time, 0.1)
 146        self.last_time = current_time
 147
 148        if getattr(self, "game_over_triggered", False):
 149            return
 150        
 151        self.update_crt()
 152        
 153        for effect in self.visual_effects[:]:
 154            if effect.update(dt):
 155                self.removeItem(effect)
 156                self.visual_effects.remove(effect)
 157        
 158        # dialogue continue meme si jeu bloque
 159        self.dialogue_manager.update(dt)
 160        
 161        if self.game_paused:
 162            return
 163
 164        if not self.is_transitioning:
 165            #gestion de surveillange du flag de magic_wall
 166            magic_flag = self.room_data.get("magic_wall")
 167            poofed_room_flag = f"magic_poofed_{self.current_room}"
 168            if magic_flag and self.check_magic_wall_condition(magic_flag) and not self.get_flag(poofed_room_flag):
 169                self.poof_all_magic_walls()
 170                
 171            self.player.update(dt, self)
 172            self._update_deferred_collisions()
 173            self.check_room_transition()
 174            self.update_animations(dt)
 175            
 176            for enemy in self.enemies:
 177                enemy.update(dt, self)
 178            self._check_pickups()    
 179            self.check_pending_npcs()                                 
 180            for interactable in self.interactables:
 181                interactable.update(dt,self)
 182        
 183        for p in self.projectiles[:]:
 184            if p.scene() is None:
 185                self.projectiles.remove(p)
 186            else:
 187                p.update(dt, self)
 188  
 189        self.transition.update(dt)
 190
 191            
 192        self.music_manager.update(dt)
 193
 194        self.hud.update_hud(self.player, self.screen_manager.inventory, self)

 309    def is_tile_blocking(self, tile_x, tile_y, entity=None):
 310        """Verifie si la case (tile_x, tile_y) de la grille est bloquante."""
 311        if not (0 <= tile_x < GRID_WIDTH and 0 <= tile_y < GRID_HEIGHT):
 312            return True
 313
 314        for dy in range(self.max_h):
 315            for dx in range(self.max_w):
 316                check_x = tile_x - dx
 317                check_y = tile_y - dy
 318
 319                if 0 <= check_x < GRID_WIDTH and 0 <= check_y < GRID_HEIGHT:
 320                    
 321                    if entity == self.player and (check_x, check_y) in self.deferred_collision_tiles:
 322                        continue
 323                    
 324                    tile_id = self.room_data["tiles"][check_y][check_x]
 325                    tile = TILE_TYPES.get(tile_id)
 326
 327                    if not tile:
 328                        continue
 329                    size_w, size_h = tile.get("size", (1, 1))
 330
 331                    if dx < size_w and dy < size_h:
 332                        
 333                        if entity:
 334                            # jesus update
 335                            if tile["name"] == "water" and getattr(entity, 'can_go_on_water', False):
 336                                continue
 337                            if tile["name"] == "black" and getattr(entity, 'can_go_on_black', False):
 338                                continue
 339
 340                        if tile.get("collision") == 1:
 341                            return True
 342                            
 343        return False

 539    def update_animations(self, dt):
 540        """Met a jour l'affichage des tuiles animees et des effets globaux (ex: neige)."""
 541        self.animation_timer += dt
 542        frame_index = int(self.animation_timer / self.frame_duree)
 543    
 544        for item, tile_id in self.animated_tile_items:
 545            frames = self.tileset[tile_id]
 546            current_frame = frame_index % len(frames)
 547            item.setPixmap(frames[current_frame])
 548            
 549        # animation overlay neige
 550        if hasattr(self, 'snow_overlay') and self.snow_overlay and self.snow_frames:
 551            snow_frame_index = int((self.animation_timer * FPS) / self.frame_duree)
 552            current_frame = snow_frame_index % len(self.snow_frames)
 553            brush = self.snow_overlay.brush()
 554            brush.setTexture(self.snow_frames[current_frame])
 555            self.snow_overlay.setBrush(brush)

 142    def game_loop(self):
 143        """Boucle principale declenchee a chaque frame (mise a jour entites, HUD, salle)."""
 144        # last_time est mis a jour en premier pour eviter un grand dt au depause
 145        current_time = time.time()
 146        dt = min(current_time - self.last_time, 0.1)
 147        self.last_time = current_time
 148
 149        if getattr(self, "game_over_triggered", False):
 150            return
 151        
 152        self.update_crt()
 153        
 154        for effect in self.visual_effects[:]:
 155            if effect.update(dt):
 156                self.removeItem(effect)
 157                self.visual_effects.remove(effect)
 158        
 159        # dialogue continue meme si jeu bloque
 160        self.dialogue_manager.update(dt)
 161        
 162        if self.game_paused:
 163            return
 164
 165        if not self.is_transitioning:
 166            #gestion de surveillange du flag de magic_wall
 167            magic_flag = self.room_data.get("magic_wall")
 168            poofed_room_flag = f"magic_poofed_{self.current_room}"
 169            if magic_flag and self.check_magic_wall_condition(magic_flag) and not self.get_flag(poofed_room_flag):
 170                self.poof_all_magic_walls()
 171                
 172            self.player.update(dt, self)
 173            self._update_deferred_collisions()
 174            self.check_room_transition()
 175            self.update_animations(dt)
 176            
 177            for enemy in self.enemies:
 178                enemy.update(dt, self)
 179            self._check_pickups()    
 180            self.check_pending_npcs()                                 
 181            for interactable in self.interactables:
 182                interactable.update(dt,self)
 183        
 184        for p in self.projectiles[:]:
 185            if p.scene() is None:
 186                self.projectiles.remove(p)
 187            else:
 188                p.update(dt, self)
 189  
 190        self.transition.update(dt)
 191
 192            
 193        self.music_manager.update(dt)
 194
 195        self.hud.update_hud(self.player, self.screen_manager.inventory, self)

 310    def is_tile_blocking(self, tile_x, tile_y, entity=None):
 311        """Verifie si la case (tile_x, tile_y) de la grille est bloquante."""
 312        if not (0 <= tile_x < GRID_WIDTH and 0 <= tile_y < GRID_HEIGHT):
 313            return True
 314
 315        for dy in range(self.max_h):
 316            for dx in range(self.max_w):
 317                check_x = tile_x - dx
 318                check_y = tile_y - dy
 319
 320                if 0 <= check_x < GRID_WIDTH and 0 <= check_y < GRID_HEIGHT:
 321                    
 322                    if entity == self.player and (check_x, check_y) in self.deferred_collision_tiles:
 323                        continue
 324                    
 325                    tile_id = self.room_data["tiles"][check_y][check_x]
 326                    tile = TILE_TYPES.get(tile_id)
 327
 328                    if not tile:
 329                        continue
 330                    size_w, size_h = tile.get("size", (1, 1))
 331
 332                    if dx < size_w and dy < size_h:
 333                        
 334                        if entity:
 335                            # jesus update
 336                            if tile["name"] == "water" and getattr(entity, 'can_go_on_water', False):
 337                                continue
 338                            if tile["name"] == "black" and getattr(entity, 'can_go_on_black', False):
 339                                continue
 340
 341                        if tile.get("collision") == 1:
 342                            return True
 343                            
 344        return False
