183    def move(self, dx, dy, dt, scene):
184        """
185        Deplace l'entite en tenant compte des collisions et de la correction de coin.
186        Details :
187            - Si self.collision est False, le deplacement est applique sans verification.
188            - La hitbox est rétrécie de MARGIN pixels sur chaque cote avant chaque test
189            de collision, pour tolérer un leger chevauchement visuel avec les murs.
190            - Si self.corner_correction est True et que l'axe est bloque, on tente
191            un leger decalage perpendiculaire (corner) dans les deux sens pour faire
192            glisser l'entite autour des coins au lieu de la bloquer net.
193            - Le nudge applique sur Y lors de la correction X est repercute sur new_y
194            pour éviter que le bloc Y ne l'écrase immédiatement après.
195        """
196        new_x = self.x + dx * self.speed * dt
197        new_y = self.y + dy * self.speed * dt
198
199        if not self.collision:
200            self.x = new_x
201            self.y = new_y
202            return
203
204        MARGIN = 6 * settings.scale/4         # Reduction de la hitbox en pixels pour les tests de collision
205        corner = settings.tile_size * 0.3  # Amplitude du decalage de correction de coin
206
207        def passable(x, y):
208            """Retourne True si la hitbox retrecie à (x, y) ne chevauche aucun mur."""
209            return not scene.is_blocking_rect(*self.shrink_hitbox(*self.get_hitbox(x, y), MARGIN),entity=self)
210
211        # --- Axe X
212        if passable(new_x, self.y):
213            self.x = new_x
214        elif self.corner_correction:
215            for nudge in (corner, -corner):
216                if passable(new_x, self.y + nudge) and passable(self.x, self.y + nudge):
217                    self.x = new_x
218                    self.y += nudge
219                    new_y += nudge  # repercuter le nudge pour ne pas ecraser la correction en Y
220                    break
221
222        # --- Axe Y
223        if passable(self.x, new_y):
224            self.y = new_y
225        elif self.corner_correction:
226            for nudge in (corner, -corner):
227                if passable(self.x + nudge, new_y) and passable(self.x + nudge, self.y):
228                    self.y = new_y
229                    self.x += nudge
230                    break

335    def get_knockback(self, scene, source=None):
336        """
337        CETTE FONCITON EST APPELEE LORSQUE C'EST L'ASSAILLANT QUI RECOIT RECUL APRES COUP!!!
338        
339        Fonction utilisee dans take_damage, mais cree ici a part car besoin de l'invoquer
340        lorsque entite qui frappe n'inflige pas de degats. 
341        Donc pouvoir recevoir recul sans take_damage
342        
343        Fonctionnement: on prend les centres de hitbox, et recule de x tiles dans vecteur oppose
344        """
345        # knockback
346        if source is not None:
347            sx, sy = source.get_center()
348            tx, ty = self.get_center()
349    
350            dx = tx - sx
351            dy = ty - sy
352    
353            dist = (dx**2 + dy**2) ** 0.5
354    
355            if dist > 0:
356                dx /= dist
357                dy /= dist
358    
359                # on utilise le knockback de l'assaillant en pixels
360                distance = source.knockback * settings.tile_size
361                
362                
363                self.kb_dir_x = dx
364                self.kb_dir_y = dy
365                
366                self.kb_remaining = distance
367                self.kb_speed = distance / source.duree_knockback
368                
369                self.kb_active = True

371    def apply_knockback(self, dt, scene):
372        """
373        CETTE FONCITON EST APPELEE LORSQUE C'EST L'ASSAILLANT QUI INFLINGE RECUL A L'ENNEMI!!!
374        
375        Fonciton qui va bouger petit a petit le joueur 
376        dans direction du knockback, en fonction de kb_speed
377        
378        On fait petit a petit, pour transition, 
379        et on bloque mouvement par _move_with_collision_limit
380        """
381        if not self.kb_active:
382            return
383    
384        move_step = self.kb_speed * dt
385    
386        # pas depasser distance totale
387        if move_step > self.kb_remaining:
388            move_step = self.kb_remaining
389    
390        dx = self.kb_dir_x * move_step
391        dy = self.kb_dir_y * move_step
392    
393        self._move_with_collision_limit("x", dx)
394        self._move_with_collision_limit("y", dy)
395    
396        self.kb_remaining -= move_step
397    
398        if self.kb_remaining <= 0:
399            self.kb_active = False

402    def _move_with_collision_limit(self, axis, amount):
403        """Deplace pixel par pixel sur un axe jusqu'a bloquer contre un mur ou une limite."""
404        if amount == 0:
405            return
406    
407        steps = int(abs(amount))
408        direction = 1 if amount > 0 else -1
409    
410        # deplacement pixel par pixelcar si d'un coup foncitonne pas
411        for _ in range(steps):
412            if axis == "x":
413                test_x = self.x + direction
414                hx, hy, hw, hh = self.get_hitbox(test_x, self.y)
415                
416                #bloquer le deplcament si collision
417                if self.scene().is_blocking_rect(hx, hy, hw, hh,self):
418                    return
419                
420                #bloquer le deplacement si oob
421                if self._is_out_of_bounds(test_x, self.y):
422                    return
423    
424                self.x = test_x
425    
426            else:
427                test_y = self.y + direction
428                hx, hy, hw, hh = self.get_hitbox(self.x, test_y)
429    
430                if self.scene().is_blocking_rect(hx, hy, hw, hh,self):
431                    return
432                
433                if self._is_out_of_bounds(self.x,test_y):
434                    return
435    
436                self.y = test_y
437
438        rest = abs(amount) - steps
439        if rest <= 0:
440            return
441        
442        # avec le reste
443        if axis == "x":
444            test_x = self.x + direction * rest
445            hx, hy, hw, hh = self.get_hitbox(test_x, self.y)
446    
447            if not self.scene().is_blocking_rect(hx, hy, hw, hh,self) and not self._is_out_of_bounds(test_x, self.y):
448                self.x = test_x
449    
450        else:
451            test_y = self.y + direction * rest
452            hx, hy, hw, hh = self.get_hitbox(self.x, test_y)
453    
454            if not self.scene().is_blocking_rect(hx, hy, hw, hh,self) and not self._is_out_of_bounds(self.x,test_y):
455                self.y = test_y

457    def _is_out_of_bounds(self, x, y):
458        """Verifie si les coordonnees de la hitbox sortent des limites de la scene."""
459        scene = self.scene()
460        if not scene:
461            return False
462    
463        w = GRID_WIDTH * settings.tile_size
464        h = (GRID_HEIGHT + HUD_HEIGHT) * settings.tile_size
465    
466        hx, hy, hw, hh = self.get_hitbox(x, y)
467    
468        return (
469            hx < 0 or
470            hy < HUD_HEIGHT * settings.tile_size or
471            hx + hw > w or
472            hy + hh > h
473        )

534                 self.effect_immunity_timer = 0

544    def apply_stun_wiggle(self, dt, scene):
545        """
546        C'est juste un truc qui fait wiggle quoi
547        """
548        if not self.is_stunned or not self.enable_stun_wiggle:
549            return
550        if self.stun_wiggle_duration <= 0:
551            return
552    
553        self.stun_wiggle_timer += dt
554    
555        t = self.stun_wiggle_timer / self.stun_wiggle_duration
556    
557        if t > 1:
558            return
559    
560        # en divise mvnt en 3 phases : 0.1 -> -0.2 -> 0.1
561        #dist_wiggle = 0.1 # parametre de distance wiggle
562        A = 0.05 # parametre de distance wiggle
563        if t < 0.33:
564            offset = (t / 0.33) * A
565        elif t < 0.66:
566            offset = A - ((t - 0.33) / 0.33) * (3*A)
567        else:
568            offset = -2*A + ((t - 0.66) / 0.34) * (3*A)
569    
570        offset_pixels = offset * settings.tile_size
571    
572        dx = self.stun_perp_x * offset_pixels
573        dy = self.stun_perp_y * offset_pixels
574        
575        # pour pas aller dans murs
576        self._move_with_collision_limit("x", dx)
577        self._move_with_collision_limit("y", dy)

184    def move(self, dx, dy, dt, scene):
185        """
186        Deplace l'entite en tenant compte des collisions et de la correction de coin.
187        Details :
188            - Si self.collision est False, le deplacement est applique sans verification.
189            - La hitbox est rétrécie de MARGIN pixels sur chaque cote avant chaque test
190            de collision, pour tolérer un leger chevauchement visuel avec les murs.
191            - Si self.corner_correction est True et que l'axe est bloque, on tente
192            un leger decalage perpendiculaire (corner) dans les deux sens pour faire
193            glisser l'entite autour des coins au lieu de la bloquer net.
194            - Le nudge applique sur Y lors de la correction X est repercute sur new_y
195            pour éviter que le bloc Y ne l'écrase immédiatement après.
196        """
197        new_x = self.x + dx * self.speed * dt
198        new_y = self.y + dy * self.speed * dt
199
200        if not self.collision:
201            self.x = new_x
202            self.y = new_y
203            return
204
205        MARGIN = 6 * settings.scale/4         # Reduction de la hitbox en pixels pour les tests de collision
206        corner = settings.tile_size * 0.3  # Amplitude du decalage de correction de coin
207
208        def passable(x, y):
209            """Retourne True si la hitbox retrecie à (x, y) ne chevauche aucun mur."""
210            return not scene.is_blocking_rect(*self.shrink_hitbox(*self.get_hitbox(x, y), MARGIN),entity=self)
211
212        # --- Axe X
213        if passable(new_x, self.y):
214            self.x = new_x
215        elif self.corner_correction:
216            for nudge in (corner, -corner):
217                if passable(new_x, self.y + nudge) and passable(self.x, self.y + nudge):
218                    self.x = new_x
219                    self.y += nudge
220                    new_y += nudge  # repercuter le nudge pour ne pas ecraser la correction en Y
221                    break
222
223        # --- Axe Y
224        if passable(self.x, new_y):
225            self.y = new_y
226        elif self.corner_correction:
227            for nudge in (corner, -corner):
228                if passable(self.x + nudge, new_y) and passable(self.x + nudge, self.y):
229                    self.y = new_y
230                    self.x += nudge
231                    break

373        """
374        CETTE FONCITON EST APPELEE LORSQUE C'EST L'ASSAILLANT QUI INFLINGE RECUL A L'ENNEMI!!!
375        
376        Fonciton qui va bouger petit a petit le joueur 
377        dans direction du knockback, en fonction de kb_speed
378        
379        On fait petit a petit, pour transition, 
380        et on bloque mouvement par _move_with_collision_limit
381        """
382        if not self.kb_active:
383            return
384    
385        move_step = self.kb_speed * dt
386    
387        # pas depasser distance totale
388        if move_step > self.kb_remaining:
389            move_step = self.kb_remaining
390    
391        dx = self.kb_dir_x * move_step
392        dy = self.kb_dir_y * move_step
393    
394        self._move_with_collision_limit("x", dx)
395        self._move_with_collision_limit("y", dy)
396    
397        self.kb_remaining -= move_step
398    
399        if self.kb_remaining <= 0:
400            self.kb_active = False

403    def _move_with_collision_limit(self, axis, amount):
404        """Deplace pixel par pixel sur un axe jusqu'a bloquer contre un mur ou une limite."""
405        if amount == 0:
406            return
407    
408        steps = int(abs(amount))
409        direction = 1 if amount > 0 else -1
410    
411        # deplacement pixel par pixelcar si d'un coup foncitonne pas
412        for _ in range(steps):
413            if axis == "x":
414                test_x = self.x + direction
415                hx, hy, hw, hh = self.get_hitbox(test_x, self.y)
416                
417                #bloquer le deplcament si collision
418                if self.scene().is_blocking_rect(hx, hy, hw, hh,self):
419                    return
420                
421                #bloquer le deplacement si oob
422                if self._is_out_of_bounds(test_x, self.y):
423                    return
424    
425                self.x = test_x
426    
427            else:
428                test_y = self.y + direction
429                hx, hy, hw, hh = self.get_hitbox(self.x, test_y)
430    
431                if self.scene().is_blocking_rect(hx, hy, hw, hh,self):
432                    return
433                
434                if self._is_out_of_bounds(self.x,test_y):
435                    return
436    
437                self.y = test_y
438
439        rest = abs(amount) - steps
440        if rest <= 0:
441            return
442        
443        # avec le reste
444        if axis == "x":
445            test_x = self.x + direction * rest
446            hx, hy, hw, hh = self.get_hitbox(test_x, self.y)
447    
448            if not self.scene().is_blocking_rect(hx, hy, hw, hh,self) and not self._is_out_of_bounds(test_x, self.y):
449                self.x = test_x
450    
451        else:
452            test_y = self.y + direction * rest
453            hx, hy, hw, hh = self.get_hitbox(self.x, test_y)
454    
455            if not self.scene().is_blocking_rect(hx, hy, hw, hh,self) and not self._is_out_of_bounds(self.x,test_y):
456                self.y = test_y

189            - La hitbox est rétrécie de MARGIN pixels sur chaque cote avant chaque test

193            glisser l'entite autour des coins au lieu de la bloquer net.

195            pour éviter que le bloc Y ne l'écrase immédiatement après.
